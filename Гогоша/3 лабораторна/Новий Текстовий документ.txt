int main(void)

    printf("Enter email: ");

    printf("Enter favorite color: ");
    scanf("%19s", color1);

    printf("\nPerson 2\n");

    printf("Enter surname: ");
    scanf("%29s", surname2);

    printf("Enter email: ");
    scanf("%49s", email2);

    printf("%-4d %-29s %-49s %-19s\n",

           2, surname2, email2, color2);
